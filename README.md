def send_all_emails():
    if not CV_PDF_PATH.is_file():
        raise FileNotFoundError(
            f"CV not found: {CV_PDF_PATH}. Put the PDF beside this script or update CV_PDF_PATH."
        )
    if not OUTREACH_LIST:
        raise ValueError("OUTREACH_LIST is empty.")

    total = len(OUTREACH_LIST)
    confirmation = input(
        f"This will send {total} individual emails with your CV. "
        "Type SEND to continue: "
    ).strip()
    if confirmation != "SEND":
        print("Cancelled; no emails were sent.")
        return

    app_password = os.environ.get("GMAIL_APP_PASSWORD", "").strip().replace(" ", "")
    if not app_password:
        app_password = getpass(f"Gmail App Password for {SENDER_EMAIL}: ").strip().replace(" ", "")
    if not app_password:
        raise ValueError("No Gmail App Password was provided.")

    cv_bytes = CV_PDF_PATH.read_bytes()
    context = ssl.create_default_context()
    sent = 0
    attempted = 0
    failed = []

    with smtplib.SMTP_SSL("smtp.gmail.com", 465, context=context, timeout=30) as server:
        try:
            server.login(SENDER_EMAIL, app_password)
        except smtplib.SMTPAuthenticationError as exc:
            raise SystemExit("Gmail login failed. Use a Gmail App Password with 2-Step Verification enabled.") from exc

        for i, item in enumerate(OUTREACH_LIST, start=1):
            attempted += 1
            msg = EmailMessage()
            msg["From"] = formataddr((SENDER_NAME, SENDER_EMAIL))
            msg["To"] = item["to"]
            msg["Reply-To"] = SENDER_EMAIL
            msg["Subject"] = item["subject"]
            msg.set_content(item["body"])
            msg.add_attachment(
                cv_bytes,
                maintype="application",
                subtype="pdf",
                filename="Nishant_Singh_CV.pdf",
            )

            try:
                refused = server.send_message(msg)
                if refused:
                    rejected = ", ".join(refused)
                    failed.append((item["company"], item["to"], f"Rejected: {rejected}"))
                    print(f"[{i}/{total}] Rejected for {item['company']} ({rejected})")
                else:
                    sent += 1
                    print(f"[{i}/{total}] Sent to {item['company']} ({item['to']})")
            except smtplib.SMTPRecipientsRefused as exc:
                failed.append((item["company"], item["to"], str(exc)))
                print(f"[{i}/{total}] Recipient refused for {item['company']}: {exc}")
            except (smtplib.SMTPException, OSError) as exc:
                failed.append((item["company"], item["to"], str(exc)))
                print(f"[{i}/{total}] SMTP error for {item['company']}: {exc}")
                print("Stopping because the mail server connection may no longer be usable.")
                break

            if i < total:
                time.sleep(2.0)

    print(f"Finished: {sent} sent, {len(failed)} failed, {total - attempted} not attempted (of {total}).")
    for company, recipient, reason in failed:
        print(f"  - {company} ({recipient}): {reason}")


if __name__ == "__main__":
    send_all_emails()
